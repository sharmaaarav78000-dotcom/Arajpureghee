        function closeCart() { cartOverlay.classList.remove('active'); cartPanel.classList.remove('active'); }
        function openCheckout() { closeCart(); checkoutOverlay.classList.add('active'); checkoutModal.classList.add('active'); }
        function closeCheckout() { checkoutOverlay.classList.remove('active'); checkoutModal.classList.remove('active'); }

        function togglePaymentView() {
            const isUPI = document.querySelector('input[name="paymentType"]:checked').value === 'UPI';
            if (isUPI) {
                qrSection.style.display = 'block';
                upiTxnId.setAttribute('required', 'true');
            } else {
                qrSection.style.display = 'none';
                upiTxnId.removeAttribute('required');
            }
        }

        function placeOrder(event) {
            event.preventDefault(); 

            const name = document.getElementById('custName').value;
            const phone = document.getElementById('custPhone').value;
            const address = document.getElementById('custAddress').value;
            const paymentType = document.querySelector('input[name="paymentType"]:checked').value;
            const txnId = upiTxnId.value;
            
            const subtotalAmt = cartQuantity * product.price;
            let totalBeforeTax = subtotalAmt - appliedDiscount;
            if(totalBeforeTax < 0) totalBeforeTax = 0;
            
            let taxAmt = Math.round(totalBeforeTax * 0.05);
            let finalTotal = totalBeforeTax + taxAmt;

            let waMessage = `*NEW ORDER - ARAJ PURE*\n\n`;
            waMessage += `*Product:* ${product.name}\n`;
            waMessage += `*Quantity:* ${cartQuantity}\n`;
            waMessage += `*MRP Subtotal:* ₹${subtotalAmt.toLocaleString('en-IN')}\n`;
            
            if(appliedDiscount > 0) {
                waMessage += `*Discount Applied:* -₹${appliedDiscount.toLocaleString('en-IN')}\n`;
            }
            
            waMessage += `*Tax (5%):* +₹${taxAmt.toLocaleString('en-IN')}\n`;
            waMessage += `*Final Amount:* ₹${finalTotal.toLocaleString('en-IN')}\n\n`;
            waMessage += `*Customer Details:*\n`;
            waMessage += `Name: ${name}\n`;
            waMessage += `Phone: ${phone}\n`;
            waMessage += `Address: ${address}\n\n`;
            
            if (paymentType === 'COD') {
                waMessage += `*Payment Method:* Cash on Delivery (COD)\n`;
            } else {
                waMessage += `*Payment Method:* UPI (Prepaid)\n`;
                waMessage += `*Transaction ID:* ${txnId}\n`;
            }

            const myWhatsAppNumber = "918979221409"; 
            const waLink = `https://wa.me/${myWhatsAppNumber}?text=${encodeURIComponent(waMessage)}`;
            window.open(waLink, '_blank');

            alert("Order Prepared! Redirecting to WhatsApp to complete your order.");
            
            cartQuantity = 0;
            appliedDiscount = 0;
            document.getElementById('couponSelect').value = "";
            couponMessage.style.display = 'none';
            couponMessage.classList.remove('success-msg-anim');
            updateCartUI();
            closeCheckout();
            document.getElementById('checkoutForm').reset();
            togglePaymentView();
        }
